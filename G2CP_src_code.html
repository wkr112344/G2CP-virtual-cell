# -*- coding: utf-8 -*-
"""_build_prot_graph.py —— 蛋白相似图传播(论文 UniPert 蛋白策略的轻量落地)。
论文: 对每个蛋白做 Smith-Waterman 局部比对构建相似图, GNN 消息传递传播功能信息。
我们: 用 ESM2-8M 向量算相似度(4994×320 余弦) → top-K 邻居图 → 消息传递 2 轮,
      esm_smoothed = esm + α·mean(邻居 esm) → 重新归一化。
输出: data/depmap/esm_cache_smoothed.pt {gene: 320维}
"""
import os, sys
import numpy as np
import torch

BASE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, BASE)

esm_cache = torch.load(os.path.join(BASE, "data", "depmap", "esm_cache.pt"),
                       map_location="cpu", weights_only=False)
genes = sorted(esm_cache.keys())
X = np.stack([esm_cache[g].astype(np.float32) for g in genes])
X = X / (np.linalg.norm(X, axis=1, keepdims=True) + 1e-8)
print(f"ESM 矩阵: {X.shape} ({len(genes)} 蛋白)", flush=True)
Page 5/60
# ---- top-K 相似邻居图 ----
K = 20
S = X @ X.T                      # 4994×4994 余弦
np.fill_diagonal(S, -1)
topk = np.argsort(-S, axis=1)[:, :K]
print(f"相似图: 每蛋白 top-{K} 邻居, 平均相似度 {S[np.arange(len(genes))[:,None], topk].mean():.3f}", flush=True)

# ---- 消息传递 2 轮(α=0.3, 论文式功能信息传播) ----
alpha = 0.3
Y = X.copy()
for it in range(2):
    neigh_mean = Y[topk].mean(axis=1)          # 邻居平均
    Y = Y + alpha * neigh_mean
    Y = Y / (np.linalg.norm(Y, axis=1, keepdims=True) + 1e-8)
    print(f"  第{it+1}轮传播: 与原始ESM平均余弦 {np.mean((Y * X).sum(1)):.3f}", flush=True)

# ---- 保存 ----
out = {g: Y[i].astype(np.float32) for i, g in enumerate(genes)}
torch.save(out, os.path.join(BASE, "data", "depmap", "esm_cache_smoothed.pt"))
print(f"已保存 esm_cache_smoothed.pt: {len(out)} 蛋白", flush=True)

def target_to_genes(target_str):
    """解析 target 注释 → 基因符号列表（去重，保持顺序）。"""
    if not target_str:
        return []
    target_str = target_str.strip()
    if target_str in TARGET2GENES:
        return list(TARGET2GENES[target_str])
    out = []
    for part in re.split(r"[,\+/]", target_str):
        part = part.strip()
        genes = TARGET2GENES.get(part, [])
        if not genes and part and part[0].isupper() and not any(
                c in part for c in " -()."):
            # 兜底：看起来像单个基因符号（如 'AURKA'）
            if re.fullmatch(r"[A-Z][A-Z0-9]{1,9}", part):
                genes = [part]
        out.extend(genes)
    seen, res = set(), []
    for g in out:
        if g and g not in seen:
            seen.add(g); res.append(g)
    return res


def build_sciplex3_cpi(chem_map, gene2idx, drug_index_by_name):
    """从 sciplex3_chems.json 构造 CPI 正对。
    返回 (pairs[(gene_idx, drug_idx)], meta[(gene, drug_name, smiles)])。
    只保留 gene2idx 里已有的基因；drug_index_by_name: {name.lower(): drug_idx}。
    """
    pairs, meta = [], []
    for name, v in chem_map.items():
        di = drug_index_by_name.get(name.lower())
        if di is None:
            continue
        for g in target_to_genes(v.get("target", "")):
            gi = gene2idx.get(g)
            if gi is not None:
                pairs.append((gi, di))
                meta.append((g, name, v.get("smiles", "")))
    return pairs, meta

def train_unipert(model, loader, epochs, desc="UniPert"):
    """通用训练循环：AMP 混合精度 + 梯度累积（4GB 显存友好）。"""
    optim = torch.optim.AdamW(
        model.parameters(), lr=LEARNING_RATE, weight_decay=WEIGHT_DECAY
    )
    scaler = torch.amp.GradScaler("cuda", enabled=(DEVICE == "cuda"))
    model.train()
    for ep in range(epochs):
        running = 0.0
        acc = 0
        optim.zero_grad(set_to_none=True)

            gene_ids = batch["gene_ids"].to(DEVICE)
            cell_lines = batch["cell_lines"].to(DEVICE)
            with torch.amp.autocast("cuda", enabled=(DEVICE == "cuda")):
                g, c = model(
                    batch["gene_seqs"], gene_ids, batch["compound_graphs"], cell_lines
                )
                loss = info_nce(g, c) / GRAD_ACCUM
            scaler.scale(loss).backward()
            running += loss.item() * GRAD_ACCUM
            acc += 1

                scaler.step(optim)
                scaler.update()
                optim.zero_grad(set_to_none=True)
                acc = 0
        # epoch 末尾把剩余累积梯度也更新一次（避免小数据集上梯度被丢弃）
        if acc > 0:
            scaler.step(optim)
            scaler.update()
            optim.zero_grad(set_to_none=True)
        print(f"[{desc}] epoch {ep+1}/{epochs}  loss={running/len(loader):.4f}")
